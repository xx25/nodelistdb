{{define "hostname_cell_simple"}} {{/* Some rows carry no hostname of their own: aggregated multi-hostname results keep only the hostname that was tested, and a node listed with a bare IP has nothing else to show. Fall back to the tested hostname, then to a resolved address, so the cell still names the host that answered instead of a blank N/A. */}} {{$label := .Hostname}} {{if not $label}}{{$label = .TestedHostname}}{{end}} {{$labelIsIP := false}} {{if not $label}} {{if .ResolvedIPv4}}{{$label = index .ResolvedIPv4 0}}{{$labelIsIP = true}} {{else if .ResolvedIPv6}}{{$label = index .ResolvedIPv6 0}}{{$labelIsIP = true}}{{end}} {{end}} {{if $label}} {{$allIPs := ""}} {{if .ResolvedIPv4}} {{range $idx, $ip := .ResolvedIPv4}}{{if $idx}}{{$allIPs = printf "%s\n%s" $allIPs $ip}}{{else}}{{$allIPs = printf "IPv4:\n%s" $ip}}{{end}}{{end}} {{end}} {{if .ResolvedIPv6}} {{if $allIPs}}{{$allIPs = printf "%s\n\nIPv6:" $allIPs}}{{else}}{{$allIPs = "IPv6:"}}{{end}} {{range $idx, $ip := .ResolvedIPv6}}{{$allIPs = printf "%s\n%s" $allIPs $ip}}{{end}} {{end}} {{if and $allIPs (not $labelIsIP)}} {{$label}} {{else if $labelIsIP}} {{$label}} {{else}} {{$label}} {{end}} {{else}} N/A {{end}} {{end}}
